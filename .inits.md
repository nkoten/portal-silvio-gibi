# .env.local

```js
// .env.local

# URL base da sua aplicação (mude para o domínio real no Vercel depois)
NEXTAUTH_URL=http://localhost:3000

# Uma string aleatória para criptografar os tokens (Gere uma string segura)
NEXTAUTH_SECRET=sua_chave_secreta_super_segura_aqui

# Cole aqui as informações que vieram no seu arquivo JSON do Google Cloud
GOOGLE_CLIENT_EMAIL=seu-bot@silvio-gibi-portal.iam.gserviceaccount.com
# Atenção: a chave privada deve ser colada com as quebras de linha (\n) exatamente como veio no JSON
GOOGLE_PRIVATE_KEY="-----BEGIN PRIVATE KEY-----\nMIIEvgIBADAN... (resto da chave aqui) ...\n-----END PRIVATE KEY-----\n"

# ID da Planilha (Fica na URL da planilha do Google Sheets, entre /d/ e /edit)
GOOGLE_SHEET_ID=seu_id_da_planilha_aqui

# cloudinary
CLOUDINARY_CLOUD_NAME=teu_cloud_name_aqui
CLOUDINARY_API_KEY=tua_api_key_aqui
CLOUDINARY_API_SECRET=teu_api_secret_aqui

# pusher
PUSHER_APP_ID=teu_app_id_aqui
NEXT_PUBLIC_PUSHER_KEY=tua_key_aqui
PUSHER_SECRET=tua_secret_aqui
NEXT_PUBLIC_PUSHER_CLUSTER=teu_cluster_aqui
```
